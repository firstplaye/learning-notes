# SQL 和 PL/SQL 的区别
SQL 是数据库查询和操作语言，而 PL/SQL 是 Oracle 在 SQL 基础上扩展出来的过程式编程语言。  
SQL：告诉数据库“我要做什么”。  
PL/SQL：告诉数据库“按照什么顺序执行、如何判断条件、如何循环、发生错误怎么办”。

# 变量定义
```
declare
SUBTYPE employee_id_type is number(10);//预定义子类型
subtype employee_age_type is number(3);

v_employee_id employee_id_type := 1001;
v_employee_age employee_age_type := 17;
begin
if v_employee_age not between 18 and 65 then
raise_application_error(-20001, '年龄必须在18到65之间');
end if;
DBMS_OUTPUT.put_line('员工编号' || v_employee_id);
dbms_output.put_line('员工年龄' || v_employee_age);
end;
/
```
声明部分（DECLARE）（可选） 定义变量、游标、常量、子程序等。  
可执行部分（BEGIN … END）（必须） 编写可执行的 PL/SQL 与 SQL 语句。  
异常处理部分（EXCEPTION）（可选） 处理运行时异常。  

计数、ID：PLS_INTEGER  
金额、精度计算：NUMBER(p,s)  
字符串统一用varchar2  

## boolean
### 为什么 BOOLEAN 有三个值？
因为数据库中存在 NULL，它表示未知，而不是 FALSE。
**Oracle 23c 之前的 BOOLEAN 使用限制,SQL 层不支持 BOOLEAN 数据类型。需要注意：从 Oracle Database 23c（后续版本包括 23ai） 开始，SQL 也支持 BOOLEAN 数据类型了。**
```
declare
v_salary number := 350000;
v_years number := 3;

v_is_active boolean := true;
v_can_get_bonus boolean;

begin
v_can_get_bonus := v_is_active and v_salary >= 300000 and v_years >= 2;
if v_can_get_bonus then
dbms_output.put_line('ok');
ELSIF not v_can_get_bonus then
dbms_output.put_line('no');
else
dbms_output.put_line('null');
end if;
end;
/
```

# SELECT INTO
```
DECLARE
    v_name EMPLOYEES.NAME%TYPE;
    v_salary EMPLOYEES.SALARY%TYPE;
BEGIN
    SELECT NAME, SALARY
    INTO v_name, v_salary
    FROM EMPLOYEES
    WHERE ID = 100;

    DBMS_OUTPUT.PUT_LINE('员工姓名：' || v_name);
    DBMS_OUTPUT.PUT_LINE('工资：' || v_salary);

EXCEPTION
    WHEN NO_DATA_FOUND THEN
        DBMS_OUTPUT.PUT_LINE('员工不存在');

    WHEN TOO_MANY_ROWS THEN
        DBMS_OUTPUT.PUT_LINE('查询到多个员工');
END;
/
```
SELECT 列数 必须与 INTO 变量数量一致,查询结果 必须且只能返回一行。

# 异常处理
异常处理（Exception Handling） 用于捕获程序运行过程中发生的错误，并执行相应的处理逻辑，避免程序因为错误而直接中断。
## 基本语法
```
DECLARE
    -- 声明变量
BEGIN
    -- 正常执行的代码

EXCEPTION
    WHEN 异常名称 THEN
        -- 异常处理代码

    WHEN OTHERS THEN
        -- 处理其他异常
END;
/
```

## WHEN OTHERS（兜底异常）
```
exception
  when others then
    dbms_output.put_line('Error code=' || sqlcode);
    dbms_output.put_line('Error msg='  || sqlerrm);
    raise;
end;
```
完整示例：
```
DECLARE
v_id EMPLOYEES.ID%TYPE := 100;
v_new_salary EMPLOYEES.salary%type := 350000;
V_old_salary EMPLOYEES.salary%type;
begin
select SALARY
into v_old_salary 
from EMPLOYEES
WHERE ID = v_id;

dbms_output.put_line('原工资' || v_old_salary);

if v_new_salary < 0 then
    raise_application_error(-20001, '工资不能小于0');
end if;

update employees
set salary = v_new_salary
where id = v_id;

commit;

dbms_output.put_line('工资修改为了:' || v_new_salary);

exception
when no_data_found then
rollback;
dbms_output.put_line('员工不存在');
when others then
rollback;
dbms_output.put_line('code：' || sqlcode);
dbms_output.put_line('message:' || sqlerrm);
raise;//将异常继续向外抛出

end;
/
```
## 用户自定义异常（业务异常）

```
DECLARE
    v_id EMPLOYEES.ID%TYPE := 100;
    v_new_salary EMPLOYEES.SALARY%TYPE := -1;
    v_old_salary EMPLOYEES.SALARY%TYPE;

    -- ① 声明用户自定义异常
    e_invalid_salary EXCEPTION;

BEGIN
    -- 查询原工资
    SELECT SALARY
    INTO v_old_salary
    FROM EMPLOYEES
    WHERE ID = v_id;

    DBMS_OUTPUT.PUT_LINE('原工资：' || v_old_salary);

        RAISE e_invalid_salary;
    END IF;

    -- 工资合法才更新
    UPDATE EMPLOYEES
    SET SALARY = v_new_salary
    WHERE ID = v_id;

EXCEPTION
    WHEN NO_DATA_FOUND THEN
        ROLLBACK;
        DBMS_OUTPUT.PUT_LINE('员工不存在');

    -- ③ 捕获自定义异常
    WHEN e_invalid_salary THEN
        ROLLBACK;
        DBMS_OUTPUT.PUT_LINE('工资不能为负数或NULL');

    WHEN OTHERS THEN
        ROLLBACK;
        DBMS_OUTPUT.PUT_LINE('其他异常：' || SQLERRM);
        RAISE;
END;
/
```

## RAISE_APPLICATION_ERROR
RAISE_APPLICATION_ERROR 是 Oracle PL/SQL 提供的一个内置过程，用来主动抛出带有错误编号和错误信息的业务异常。
```
SET SERVEROUTPUT ON;

DECLARE
    v_id EMPLOYEES.ID%TYPE := 100;
    v_new_salary EMPLOYEES.SALARY%TYPE := -1;
    v_old_salary EMPLOYEES.SALARY%TYPE;

BEGIN
    SELECT SALARY
    INTO v_old_salary
    FROM EMPLOYEES
    WHERE ID = v_id;

    DBMS_OUTPUT.PUT_LINE('原工资：' || v_old_salary);

    -- 检查新工资是否合法
    IF v_new_salary IS NULL OR v_new_salary < 0 THEN
        RAISE_APPLICATION_ERROR(
            -20001,
            '工资不能为负数或NULL'
        );
    END IF;

    UPDATE EMPLOYEES
    SET SALARY = v_new_salary
    WHERE ID = v_id;

    COMMIT;

    DBMS_OUTPUT.PUT_LINE('工资修改成功');

EXCEPTION
    WHEN NO_DATA_FOUND THEN
        DBMS_OUTPUT.PUT_LINE('员工不存在');

    WHEN OTHERS THEN
        DBMS_OUTPUT.PUT_LINE('错误编号：' || SQLCODE);
        DBMS_OUTPUT.PUT_LINE('错误信息：' || SQLERRM);
        RAISE;
END;
/
```
## 日志与自治事务
业务执行失败后需要 ROLLBACK，但我们希望错误日志仍然保存在数据库中。
### 什么是自治事务？
自治事务是 Oracle 提供的一种独立事务机制。它允许一个 PL/SQL 子程序在调用者事务尚未结束时，独立执行 COMMIT 或 ROLLBACK。
自治事务的提交不会提交主事务；主事务的回滚也不会撤销已经提交的自治事务。
#### 第一步：创建日志表
```
CREATE TABLE ERROR_LOG (
    LOG_ID NUMBER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    ERROR_CODE NUMBER,
    ERROR_MESSAGE VARCHAR2(2000),
    CREATED_AT TIMESTAMP DEFAULT SYSTIMESTAMP
);
```
#### 第二步：创建自治事务日志过程
```
CREATE OR REPLACE PROCEDURE WRITE_ERROR_LOG (//创建存储过程
    p_error_code IN NUMBER,
    p_error_message IN VARCHAR2
)
AS
    PRAGMA AUTONOMOUS_TRANSACTION;//声明该过程使用自治事务
BEGIN
    INSERT INTO ERROR_LOG (
        ERROR_CODE,
        ERROR_MESSAGE
    )
    VALUES (
        p_error_code,
        p_error_message
    );

    COMMIT;

EXCEPTION
    WHEN OTHERS THEN
        ROLLBACK;
        RAISE;
END;
/
```
#### 第三步：执行工资更新业务
```
SET SERVEROUTPUT ON;

DECLARE
    v_id EMPLOYEES.ID%TYPE := 100;
    v_new_salary EMPLOYEES.SALARY%TYPE := 500000;

    v_error_code NUMBER;
    v_error_message VARCHAR2(2000);

BEGIN
    -- ① 更新工资
    UPDATE EMPLOYEES
    SET SALARY = v_new_salary
    WHERE ID = v_id;

    IF SQL%ROWCOUNT = 0 THEN
        RAISE_APPLICATION_ERROR(-20002, '员工不存在');
    END IF;

    DBMS_OUTPUT.PUT_LINE('工资已更新，但尚未提交');

    -- ② 模拟业务错误
    RAISE_APPLICATION_ERROR(-20001, '模拟工资处理失败');

    COMMIT;

EXCEPTION
    WHEN OTHERS THEN
        -- ③ 保存异常信息
        v_error_code := SQLCODE;
        v_error_message := SQLERRM;

        -- ④ 调用自治事务记录日志
        WRITE_ERROR_LOG(
            v_error_code,
            v_error_message
        );

        -- ⑤ 回滚主事务
        ROLLBACK;

        DBMS_OUTPUT.PUT_LINE('工资修改已回滚');
        DBMS_OUTPUT.PUT_LINE('错误日志已保存');
END;
/
```
### 自治事务的重要注意事项 
自治事务执行了 DML 后，必须通过 COMMIT 或 ROLLBACK 结束事务，才能正常返回调用者。  
日志失败也需要考虑

# PL/SQL 存储过程与函数
在 Oracle PL/SQL 中，存储过程（Procedure）和函数（Function）都是可以保存在数据库中、反复调用的程序单元。
## 存储过程与函数的区别
![alt text](image-2.png)
函数 不能执行 DML（INSERT/UPDATE/DELETE）  
函数不能 COMMIT / ROLLBACK  
函数应是 确定性（无副作用）  
## 示例代码
存储过程procedure
```
create or replace NONEDITIONABLE procedure get_employee_info(
    p_id     in  employees.id%type,
    p_name   out employees.name%type,
    p_salary out employees.salary%type)
as
begin
    select name,salary
    into p_name, p_salary
    from employees
    where id = p_id;

exception
    when no_data_found then
        raise_application_error(-20001, '员工不存在');
    when others THEN
        dbms_output.put_line('code:' || sqlcode);
        dbms_output.put_line('message' || sqlerrm);
    raise;
end;
```
存储函数function
```
create or replace NONEDITIONABLE FUNCTION get_annual_salary(
    p_id IN employees.id%TYPE
)
RETURN NUMBER
AS
    v_salary employees.salary%TYPE;
BEGIN
    -- 查询员工工资
    SELECT salary
    INTO v_salary
    FROM employees
    WHERE id = p_id;

    -- 返回年收入
    RETURN v_salary * 12;

EXCEPTION
    WHEN NO_DATA_FOUND THEN
        RAISE_APPLICATION_ERROR(-20001, '员工不存在');
END;
```

# 游标(Cursor)
主要用于处理 SQL 查询返回的多行数据，让 PL/SQL 能够逐行读取和处理。  
##  隐式游标（Implicit Cursor）
当你在 PL/SQL 中执行以下 SQL 时，Oracle 会自动管理相关的隐式游标：INSERT,UPDATE,DELETE,MERGE,SELECT ... INTO
### 隐式游标的四个属性
![alt text](image-3.png)
```
BEGIN
    UPDATE employees
    SET salary = salary + 10000
    WHERE department = 'HR';

    DBMS_OUTPUT.PUT_LINE(
        '修改了 ' || SQL%ROWCOUNT || ' 行'
    );

    ROLLBACK;
END;
/
```
## 显式游标（Explicit Cursor）
显式游标（明示カーソル）是程序员自己声明、打开、读取和关闭的游标。当 SQL 查询返回多行数据时，在 PL/SQL 中逐行读取并处理这些数据。
```
SET SERVEROUTPUT ON;

declare
cursor c_emp is
select id, name, salary
from employees
order by id;

v_emp c_emp%ROWTYPE;

begin
    open c_emp;
    loop
        fetch c_emp into v_emp;
        exit when c_emp%notfound;
        dbms_output.put_line(
            'id:' || v_emp.id || CHR(10) ||
            'name:' || v_emp.name || CHR(10) ||
            'salary' || v_emp.salary);
    end loop;
    
    close c_emp;
end;
/
```

## 游标 FOR 循环
游标 FOR 循环是 最简洁、最安全、最常用 的方式：自动 OPEN / FETCH / CLOSE,不易写错。
```
SET SERVEROUTPUT ON;

BEGIN
    FOR v_emp IN (
        SELECT id, name, salary
        FROM employees
        ORDER BY id
    )
    LOOP
        IF v_emp.salary >= 400000 THEN
            DBMS_OUTPUT.PUT_LINE(
                v_emp.name || '：高级'
            );

        ELSIF v_emp.salary >= 300000 THEN
            DBMS_OUTPUT.PUT_LINE(
                v_emp.name || '：中级'
            );

        ELSE
            DBMS_OUTPUT.PUT_LINE(
                v_emp.name || '：初级'
            );
        END IF;
    END LOOP;
END;
/
```

# PL/SQL 触发器
触发器（Trigger / トリガー）是一种保存在数据库中的程序，当指定事件发生时，Oracle 会自动执行它。触发器最大的特点：不需要手动调用，而是由数据库事件自动触发。
## 1. 行级触发器（Row-Level Trigger）
```
create or replace NONEDITIONABLE trigger trg_check_salary
before insert or update of salary
on employees
for each row
begin
    if :new.salary < 0 then
        raise_application_error(
            -20001,
            '工资不能小于0'
            );
    end if;
end;
```
### :OLD 和 :NEW 是什么？
这是行级触发器中最常用的两个特殊记录。  
- :OLD：修改前的数据。  
- :NEW：修改后的新数据。  
## 2. 语句级触发器（Statement-Level Trigger）
```
CREATE OR REPLACE TRIGGER trg_employee_update
AFTER UPDATE ON employees
BEGIN
    DBMS_OUTPUT.PUT_LINE('UPDATE执行了一次');
END;
/
```

# PL/SQL 包（PACKAGE）
在 Oracle PL/SQL 中，包（PACKAGE / パッケージ）是将多个相关的存储过程、函数、变量、游标等集中管理的数据库对象。
## PACKAGE 由两个部分组成
![alt text](image-4.png)
## 完整示例：员工工资管理 PACKAGE
### 第一步：创建 PACKAGE Specification（接口类）
```
CREATE OR REPLACE PACKAGE employee_pkg AS

    -- 查询员工工资
    FUNCTION get_salary(
        p_id IN employees.id%TYPE
    ) RETURN NUMBER;

    -- 修改员工工资
    PROCEDURE update_salary(
        p_id IN employees.id%TYPE,
        p_salary IN employees.salary%TYPE
    );

    -- 计算员工年薪
    FUNCTION get_annual_salary(
        p_id IN employees.id%TYPE
    ) RETURN NUMBER;

END employee_pkg;
/
```
### 第二步：创建 PACKAGE BODY（具体业务逻辑）
```
CREATE OR REPLACE PACKAGE BODY employee_pkg AS

    -- 1. 查询员工工资
    FUNCTION get_salary(
        p_id IN employees.id%TYPE
    ) RETURN NUMBER
    AS
        v_salary employees.salary%TYPE;
    BEGIN
        SELECT salary
        INTO v_salary
        FROM employees
        WHERE id = p_id;

        RETURN v_salary;

    EXCEPTION
        WHEN NO_DATA_FOUND THEN
            RAISE_APPLICATION_ERROR(
                -20001, '员工不存在'
            );
    END get_salary;


    -- 2. 修改员工工资
    PROCEDURE update_salary(
        p_id IN employees.id%TYPE,
        p_salary IN employees.salary%TYPE
    )
    AS
    BEGIN
        IF p_salary IS NULL OR p_salary < 0 THEN
            RAISE_APPLICATION_ERROR(
                -20002, '工资不能为空或小于0'
            );
        END IF;

        UPDATE employees
        SET salary = p_salary
        WHERE id = p_id;

        IF SQL%ROWCOUNT = 0 THEN
            RAISE_APPLICATION_ERROR(
                -20003, '员工不存在'
            );
        END IF;

    END update_salary;


    -- 3. 计算员工年薪
    FUNCTION get_annual_salary(
        p_id IN employees.id%TYPE
    ) RETURN NUMBER
    AS
    BEGIN
        RETURN get_salary(p_id) * 12;
    END get_annual_salary;

END employee_pkg;
/
```
## PACKAGE 的 PUBLIC 和 PRIVATE
在包规范中声明的成员是公开的；只在包体中声明的成员是私有的。
