# PL/SQL 批量处理
批量处理（Bulk Processing / 一括処理）是指一次处理多条数据，减少 PL/SQL 与 SQL 引擎之间的交互，从而提高程序执行效率。
## 为什么普通 FOR 循环可能比较慢？
```
BEGIN
    FOR v_emp IN (
        SELECT id, salary
        FROM employees
        WHERE department = 'IT'
    )
    LOOP
        UPDATE employees
        SET salary = v_emp.salary + 1000
        WHERE id = v_emp.id;
    END LOOP;
END;
/
```
如果查询出 10 万条数据，就需要执行 10 万次单独的 UPDATE。每次执行 SQL，PL/SQL 引擎都需要将 SQL 操作交给 SQL 引擎处理，这涉及上下文切换（Context Switch / コンテキストスイッチ）。
**如果所有数据都能用一条 SQL 修改，就优先使用一条 SQL，而不是 BULK COLLECT + FORALL。**
## BULK COLLECT：批量查询
普通游标使用 FETCH 一次读取一行。而 BULK COLLECT 可以一次将多行数据放入集合。
```
SET SERVEROUTPUT ON;

DECLARE
    TYPE employee_table IS TABLE OF employees%ROWTYPE;

    v_employees employee_table;
BEGIN
    SELECT *
    BULK COLLECT INTO v_employees//它将查询结果放入 v_employees 集合。
    FROM employees
    WHERE department = 'IT';

    FOR i IN 1 .. v_employees.COUNT LOOP
        DBMS_OUTPUT.PUT_LINE(
            v_employees(i).id || ' - ' ||
            v_employees(i).name
        );
    END LOOP;
END;
/
```
## LIMIT：分批读取数据
假设有 100000 条员工数据。一次性读取全部数据可能占用大量内存。更合适的做法是每次读取 1000 条。
```
SET SERVEROUTPUT ON;

DECLARE
    CURSOR c_emp IS
        SELECT id, name, salary
        FROM employees
        ORDER BY id;

    TYPE employee_table IS TABLE OF c_emp%ROWTYPE;

    v_employees employee_table;
BEGIN
    OPEN c_emp;

    LOOP
        FETCH c_emp
        BULK COLLECT INTO v_employees
        LIMIT 10;

        EXIT WHEN v_employees.COUNT = 0;

        DBMS_OUTPUT.PUT_LINE(
            '本批读取：' || v_employees.COUNT || '条'
        );

        FOR i IN 1 .. v_employees.COUNT LOOP
            DBMS_OUTPUT.PUT_LINE(
                v_employees(i).name
            );
        END LOOP;
    END LOOP;

    CLOSE c_emp;
END;
/
```
## FORALL：批量执行 INSERT、UPDATE、DELETE
```
declare
    type id_table is table of employees.id%type;
    v_ids id_table :=  id_table(100,99,98);
begin
    forall i in 1 .. v_ids.count
        update employees
        set salary = salary + 1000
        where id = v_ids(i);
        
    dbms_output.put_line(
        '更新总行数: ' || sql%rowcount 
    );
end;
/
```
FORALL 不代表 Oracle 只执行一条 UPDATE，也不代表自动提交。 它主要通过批量绑定减少 PL/SQL 与 SQL 引擎之间的交互。

## BULK COLLECT + FORALL 组合
```
SET SERVEROUTPUT ON;

DECLARE
    -- 1. 定义员工记录类型
    TYPE t_emp IS RECORD (
        id     employees.id%TYPE,
        salary employees.salary%TYPE
    );

    -- 2. 定义集合类型
    TYPE t_emp_tab IS TABLE OF t_emp;

    -- 3. 声明集合变量
    v_emps t_emp_tab;

BEGIN
    -- 4. 批量查询员工
    SELECT id, salary
    BULK COLLECT INTO v_emps
    FROM employees
    WHERE department = 'HR';

    -- 5. 批量更新工资
    FORALL i IN 1 .. v_emps.COUNT
        UPDATE employees
        SET salary = ROUND(v_emps(i).salary * 1.05, 2)
        WHERE id = v_emps(i).id;

    -- 6. 输出更新数量
    DBMS_OUTPUT.PUT_LINE(
        '更新员工数量：' || SQL%ROWCOUNT
    );

END;
/
```

# 批处理
在指定时间，对大量数据进行集中、自动化处理的程序。

## 批处理中的事务设计
假设需要更新 10000 名员工，更新到第 6000 名时发生错误，前面已经更新的数据应该怎么办？
### 方案一：整个批处理只 COMMIT 一次
最容易保证数据一致性的设计。
```
SET SERVEROUTPUT ON;

DECLARE
    TYPE t_emp IS RECORD (
        id     employees.id%TYPE,
        salary employees.salary%TYPE
    );

    TYPE t_emp_tab IS TABLE OF t_emp;

    v_emps t_emp_tab;

BEGIN
    -- 1. 批量查询
    SELECT id, salary
    BULK COLLECT INTO v_emps
    FROM employees
    WHERE department = 'IT';

    -- 2. 批量更新
    IF v_emps.COUNT > 0 THEN

        FORALL i IN 1 .. v_emps.COUNT
            UPDATE employees
            SET salary = ROUND(v_emps(i).salary * 1.05, 2)
            WHERE id = v_emps(i).id;

        DBMS_OUTPUT.PUT_LINE(
            '更新行数：' || SQL%ROWCOUNT
        );

    END IF;

    -- 3. 全部成功后提交
    COMMIT;

    DBMS_OUTPUT.PUT_LINE('批处理成功');

EXCEPTION
    WHEN OTHERS THEN

        -- 发生异常，撤销未提交的修改
        ROLLBACK;

        DBMS_OUTPUT.PUT_LINE(
            '批处理失败：' || SQLERRM
        );

        RAISE;
END;
/
```
### 方案二：每 500 条提交一次
如果把全部数据放在一个事务中，可能出现事务持续时间过长、UNDO 占用增加、行锁保持时间过长等问题。
```
set serveroutput on;

declare 
    type t_emp is record(
        id employees.id%type,
        salary employees.salary%type
    );
    
    type t_emp_tab is table of t_emp;
    v_emps t_emp_tab;
    v_start PLS_INTEGER := 1;
    v_end PLS_INTEGER;
    
    v_batch_count number;
    v_total number := 0;
begin
    select id,salary
    BULK COLLECT INTO v_emps
    from employees
    where department = 'HR';
    
    while v_start <= v_emps.count loop
        
        v_end := LEAST(v_start + 5, v_emps.count);
        
        forall i in v_start .. v_end
            update employees
            set salary = round(v_emps(i).salary * 1.05, 2)
            where id = v_emps(i).id;
            
        v_batch_count := sql%rowcount;
        
        commit;
        
        v_total := v_total + v_batch_count;
        
        dbms_output.put_line('本地提交' || v_batch_count);
        
        v_start := v_end + 1;
        
    end loop;
    
    dbms_output.put_line('总提交行数:'||v_total);
    
    EXCEPTION
    WHEN OTHERS THEN
        ROLLBACK;
        DBMS_OUTPUT.PUT_LINE('批处理失败：' || SQLERRM);
        RAISE;
END;
/
```

## 批处理的幂等性设计
在 Oracle PL/SQL 批处理中，幂等性（Idempotency / 冪等性）指同一个业务操作执行一次和重复执行多次，最终产生的业务结果相同。
### 为什么批处理需要幂等性？
更新 500 名员工后执行 COMMIT。更新第 800 名员工时发生异常。运维人员重新执行整个批处理。前 500 名员工可能被重复加薪。
## 实现方法
### 方案一：使用处理状态控制幂等性
```
SET SERVEROUTPUT ON;

DECLARE
    -- 1. 定义员工记录类型
    TYPE t_emp IS RECORD (
        id     employee_batch_test.id%TYPE,
        salary employee_batch_test.salary%TYPE
    );

    -- 2. 定义集合类型
    TYPE t_emp_tab IS TABLE OF t_emp;

    -- 3. 声明集合变量
    v_emps t_emp_tab;

    v_count NUMBER := 0;

BEGIN
    -- 4. 只读取未处理的 HR 员工
    SELECT id, salary
    BULK COLLECT INTO v_emps
    FROM employee_batch_test
    WHERE department = 'HR'
      AND status = 'PENDING'
    FOR UPDATE;

    -- 5. 批量更新工资和状态
    IF v_emps.COUNT > 0 THEN

        FORALL i IN 1 .. v_emps.COUNT
            UPDATE employee_batch_test
            SET salary = ROUND(v_emps(i).salary * 1.05, 2),
                status = 'PROCESSED'
            WHERE id = v_emps(i).id
              AND status = 'PENDING';

        v_count := SQL%ROWCOUNT;

    END IF;

    -- 6. 提交事务
    COMMIT;

    DBMS_OUTPUT.PUT_LINE(
        '本次处理员工数：' || v_count
    );

EXCEPTION
    WHEN OTHERS THEN
        ROLLBACK;

        DBMS_OUTPUT.PUT_LINE(
            '批处理失败：' || SQLERRM
        );

        RAISE;
END;
/
```
### 方案二：使用批处理 ID 防止重复执行
给每次业务批处理分配一个唯一的业务 ID，并记录它是否执行成功。
```
SET SERVEROUTPUT ON;

DECLARE
    -- 1. 本次批处理ID
    v_batch_id VARCHAR2(50) := 'SALARY_202611';

    -- 2. 定义员工记录类型
    TYPE t_emp IS RECORD (
        id     employee_batch_test.id%TYPE,
        salary employee_batch_test.salary%TYPE
    );

    -- 3. 定义员工集合类型
    TYPE t_emp_tab IS TABLE OF t_emp;

    -- 4. 声明集合变量
    v_emps t_emp_tab;

    v_count NUMBER := 0;

BEGIN
    -- 5. 注册批次
    INSERT INTO batch_job_log (
        batch_id,
        status,
        processed_count
    )
    VALUES (
        v_batch_id,
        'RUNNING',
        0
    );

    -- 6. 批量查询员工
    SELECT id, salary
    BULK COLLECT INTO v_emps
    FROM employee_batch_test
    WHERE department = 'HR';

    -- 7. 批量更新工资
    IF v_emps.COUNT > 0 THEN

        FORALL i IN 1 .. v_emps.COUNT
            UPDATE employee_batch_test
            SET salary = ROUND(v_emps(i).salary * 1.05, 2)
            WHERE id = v_emps(i).id;

        v_count := SQL%ROWCOUNT;

    END IF;

    -- 8. 更新批处理状态
    UPDATE batch_job_log
    SET status = 'COMPLETED',
        processed_count = v_count,
        completed_at = SYSTIMESTAMP
    WHERE batch_id = v_batch_id;

    -- 9. 统一提交事务
    COMMIT;

    DBMS_OUTPUT.PUT_LINE(
        '批处理成功，更新员工数：' || v_count
    );

EXCEPTION
    WHEN DUP_VAL_ON_INDEX THEN
        -- 同一个批次已经存在
        DBMS_OUTPUT.PUT_LINE(
            '批次 ' || v_batch_id ||
            ' 已存在，跳过执行'
        );

    WHEN OTHERS THEN
        ROLLBACK;

        DBMS_OUTPUT.PUT_LINE(
            '批处理失败：' || SQLERRM
        );

        RAISE;
END;
/
```
